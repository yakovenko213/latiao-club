🎯 CRM Module Setup Helper

Quick setup scripts for your Kanban board

1 Initialize CRM Columns

Create default columns for your sales funnel. Run this in the browser console on your Admin Panel.

(async function setupCRMColumns() { if (!window.firebaseInit?.db) { alert('❌ Please run this on the admin panel with Firebase initialized'); return; } const { db, collection, setDoc, doc } = window.firebaseInit; const columns = [ { id: 'new_lead', name: 'Новий лід', description: 'Нові покинуті кошики', color: '#3b82f6', order: 0 }, { id: 'contacted', name: 'Контакт встановлено', description: 'Зв\'язалися з клієнтом', color: '#8b5cf6', order: 1 }, { id: 'negotiation', name: 'Переговори', description: 'Активні переговори', color: '#f59e0b', order: 2 }, { id: 'ready_to_order', name: 'Готовий замовити', description: 'Клієнт готовий', color: '#06b6d4', order: 3 }, { id: 'converted', name: 'Конверсія', description: 'Замовлення створено', color: '#10b981', order: 4 } ]; for (const col of columns) { await setDoc(doc(collection(db, 'crm_columns'), col.id), { name: col.name, description: col.description, color: col.color, order: col.order, created_at: new Date() }); console.log(`✅ ${col.name}`); } alert('✅ Setup complete! Navigate to: Незавершені замовлення → CRM'); })();

2 Assign Carts to Columns

Add crm_status to existing abandoned carts to populate the board.

(async function assignCartsToColumns() { if (!window.firebaseInit?.db) { alert('❌ Please run this on the admin panel'); return; } const { db, collection, getDocs, updateDoc, doc } = window.firebaseInit; const cartsRef = collection(db, 'abandoned_carts'); const snapshot = await getDocs(cartsRef); const columnIds = ['new_lead', 'contacted', 'negotiation', 'ready_to_order', 'converted']; let count = 0; for (const cartDoc of snapshot.docs) { const randomColumn = columnIds[Math.floor(Math.random() * columnIds.length)]; await updateDoc(doc(db, 'abandoned_carts', cartDoc.id), { crm_status: randomColumn }); count++; } alert(`✅ Assigned ${count} carts to CRM columns. Refresh the CRM tab!`); })();

3 Get Column Statistics

View statistics for each column (cards count and total value).

(async function getColumnStats() { if (!window.firebaseInit?.db) { alert('❌ Please run this on the admin panel'); return; } const { db, collection, getDocs, query, where } = window.firebaseInit; const columnsSnapshot = await getDocs(collection(db, 'crm_columns')); console.log('📊 CRM Column Statistics:'); console.log('═'.repeat(50)); for (const col of columnsSnapshot.docs) { const colData = col.data(); const cartsQuery = query( collection(db, 'abandoned_carts'), where('crm_status', '==', col.id) ); const cartsSnapshot = await getDocs(cartsQuery); const total = cartsSnapshot.docs.reduce((sum, cart) => { return sum + (parseFloat(cart.data().total) || 0); }, 0); console.log(`${colData.name}: ${cartsSnapshot.size} cards | ${total.toFixed(0)} ₴`); } console.log('═'.repeat(50)); alert('✅ Check console for detailed statistics'); })();

4 Calculate Conversion Rate

Calculate the conversion rate from abandoned carts to orders.

(async function getConversionRate() { if (!window.firebaseInit?.db) { alert('❌ Please run this on the admin panel'); return; } const { db, collection, getDocs, query, where } = window.firebaseInit; const allCartsQuery = query( collection(db, 'abandoned_carts'), where('crm_status', '!=', null) ); const allCarts = await getDocs(allCartsQuery); const convertedQuery = query( collection(db, 'abandoned_carts'), where('crm_status', '==', 'converted') ); const converted = await getDocs(convertedQuery); const rate = (converted.size / allCarts.size) * 100; console.log('📊 Conversion Rate Analysis:'); console.log('═'.repeat(50)); console.log(`Total Carts in CRM: ${allCarts.size}`); console.log(`Converted: ${converted.size}`); console.log(`Conversion Rate: ${rate.toFixed(1)}%`); console.log('═'.repeat(50)); alert(`📊 Conversion Rate: ${rate.toFixed(1)}%\nTotal: ${allCarts.size} | Converted: ${converted.size}`); })();
